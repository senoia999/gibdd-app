          document.getElementById('l3').textContent = final.l3;

          curPlate = `${final.l1} ${final.d1}${final.d2}${final.d3} ${final.l2}${final.l3} | ${final.reg}`;
          const evalRes = evaluatePlate(final.l1, final.d1, final.d2, final.d3, final.l2, final.l3);
          curPrice = evalRes.price;

          document.getElementById('ui-rarity-title').textContent = evalRes.name;
          document.getElementById('ui-rarity-price').textContent = `Оценка перекупа: ~${evalRes.price.toLocaleString()} ₽`;
          document.getElementById('ui-rarity-stars').textContent = evalRes.stars;

          if (evalRes.price >= 500000) {
            if (tg?.HapticFeedback) tg.HapticFeedback.notificationOccurred('success');
          }

          spinBtn.disabled = false;
          sellBtn.disabled = false;
          equipBtn.disabled = false;
        }
      }, 75);
    });

    // Продажа
    sellBtn.addEventListener('click', () => {
      if (!curPlate) return;
      balance += curPrice;
      if (tg) tg.showAlert(`🤝 Перекуп забрал номер за ${curPrice.toLocaleString()} ₽!`);
      curPlate = null;
      sellBtn.disabled = true;
      equipBtn.disabled = true;
      document.getElementById('ui-rarity-title').textContent = 'Продано перекупу!';
      document.getElementById('ui-rarity-price').textContent = 'Деньги в кармане 💰';
      updateUI();
    });

    // Установка
    equipBtn.addEventListener('click', () => {
      if (!curPlate) return;
      installedPlate = curPlate;
      if (tg) tg.showAlert(`✅ Номер ${installedPlate} теперь на вашей машине!`);
      equipBtn.disabled = true;
      updateUI();
    });

    // 🚨 Выезд в город (Рейд ДПС)
    document.getElementById('drive-city-btn').addEventListener('click', () => {
      if (installedPlate === 'ТРАНЗИТЫ') {
        balance = Math.max(0, balance - 5000);
        if (tg) tg.showAlert('🚨 ВАС ОСТАНОВИЛ ЭКИПАЖ ДПС!\n\n«Езда без номеров! Штраф 5 000 ₽ выписан по всей строгости!»');
        updateUI();
        return;
      }

      if (installedPlate.includes('ЕКХ') || installedPlate.includes('АМР')) {
        const bonus = 10000;
        balance += bonus;
        if (tg) tg.showAlert('🫡 ВСТРЕЧА С ДПС:\n\nИнспектор вытянулся по струнке, отдал воинское приветствие и перекрыл поток! За представительский класс начислено +10 000 ₽!');
        updateUI();
        return;
      }

      if (installedPlate.includes('777') || installedPlate.includes('ВОР')) {
        if (tg) tg.showAlert('🤝 ВСТРЕЧА С ДПС:\n\n«Счастливого пути, уважаемый водитель! Не нарушайте скоростной режим!»');
        return;
      }

      if (tg) tg.showAlert('🚔 ВСТРЕЧА С ДПС:\n\n«Старший лейтенант Смирнов. Документы в порядке, страховка есть. Езжайте дальше!»');
    });

    // Модалка СТС
    document.getElementById('open-sts-btn').addEventListener('click', () => {
      const car = CARS.find(c => c.id === currentCarId);
      document.getElementById('sts-plate').textContent = installedPlate;
      document.getElementById('sts-car').textContent = car.name;
      document.getElementById('sts-modal').style.display = 'flex';
    });

    // Модалка рамок
    document.getElementById('open-frame-btn').addEventListener('click', () => {
      const list = document.getElementById('frames-list');
      list.innerHTML = '';
      FRAMES.forEach(frame => {
        const b = document.createElement('button');
        b.className = 'btn btn-action';
        b.textContent = frame;
        b.onclick = () => {
          selectedFrame = frame;
          closeModal('frame-modal');
          updateUI();
        };
        list.appendChild(b);
      });
      document.getElementById('frame-modal').style.display = 'flex';
    });

    // Модалка магазина
    document.getElementById('open-shop-btn').addEventListener('click', () => {
      const list = document.getElementById('shop-cars-list');
      list.innerHTML = '';
      CARS.forEach(car => {
        const owned = ownedCars.includes(car.id);
        const cur = currentCarId === car.id;
        const row = document.createElement('div');
        row.style.cssText = 'display:flex; justify-content:space-between; align-items:center; padding:10px; background:rgba(255,255,255,0.04); margin-bottom:8px; border-radius:10px;';
        row.innerHTML = `
          <div><b>${car.emoji} ${car.name}</b><div style="font-size:12px; color:var(--text-muted);">${car.price === 0 ? 'Базовая' : car.price.toLocaleString() + ' ₽'}</div></div>
          <div>${cur ? '<span style="color:var(--accent);">Выбрана</span>' : owned ? `<button class="btn-action" style="padding:6px 12px; border-radius:8px;" onclick="chooseCar('${car.id}')">Сесть</button>` : `<button class="btn-action" style="padding:6px 12px; border-radius:8px; background:var(--green); color:#fff;" onclick="buyCar('${car.id}',${car.price})">Купить</button>`}</div>
        `;
        list.appendChild(row);
      });
      document.getElementById('shop-modal').style.display = 'flex';
    });

    window.closeModal = (id) => document.getElementById(id).style.display = 'none';
    window.chooseCar = (id) => { currentCarId = id; closeModal('shop-modal'); updateUI(); };
    window.buyCar = (id, p) => {
      if (balance < p) { if (tg) tg.showAlert('Не хватает денег!'); return; }
      balance -= p; ownedCars.push(id); currentCarId = id; closeModal('shop-modal'); updateUI();
    };

    document.getElementById('loan-btn').addEventListener('click', () => { balance += 50000; updateUI(); });

    updateUI();
  </script>
